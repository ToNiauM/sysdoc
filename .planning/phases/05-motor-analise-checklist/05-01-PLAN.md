sysdoc_dvi/tipos.py (a criar):
```python
Aplicabilidade = Literal["geral", "tic", "ambos"]
Peca = Literal["ETP", "TR", "MCP", "RISCOS", "PROCESSO"]
TipoVerificacao = Literal["presenca", "julgamento", "calculo", "vigencia", "mecanica"]
ModeloAlvo = Literal["economico", "forte"]
OrigemVerificacao = Literal["llm", "deterministico", "lookup"]

@dataclass(frozen=True)
class ItemChecklist:
    id: str                     # DVI-GER-001 | DVI-TIC-001 | DVI-AMB-001
    aplicabilidade: Aplicabilidade
    peca: Peca
    documento_saida: str        # "ETP" | "TR" (destino no dados_consolidados.json)
    tipo: TipoVerificacao
    pergunta: str
    base_legal_texto: str
    referencia: str
    severidade_default: str     # crítica | alta | média | baixa | informativa
    modelo_alvo: ModeloAlvo
    ativo: bool
    parametros: dict            # livre por tipo (ex.: tolerancia, termos, norma)

@dataclass
class UsoModelo:
    provedor: str
    modelo: str
    tokens_entrada: int
    tokens_saida: int
    custo_estimado: float
    duracao_s: float

@dataclass
class ResultadoItem:
    checklist_id: str
    origem_verificacao: OrigemVerificacao
    documento: str              # ETP | TR
    seção: str
    de: str
    para: str
    parecer: str
    fundamento: str
    classificação: str          # conforme | ajuste_necessário | risco | pendente
    severidade: str
    risco_jurídico: str
    uso: UsoModelo | None = None
    falha_estruturada: str | None = None
    falha_grounding: bool = False

class AnalysisAgent(Protocol):
    def executar_item(self, item: ItemChecklist, contexto: str) -> ResultadoItem: ...
    def executar_lote(self, itens: list[ItemChecklist], contexto: str) -> list[ResultadoItem]: ...
```
